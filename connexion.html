document.addEventListener('DOMContentLoaded', async () => { const currentUser = await window.Freshnet.getCurrentUser(); if (!currentUser) { window.location.href = 'connexion.html'; return; } const whoami = document.getElementById('whoami'); const logoutButton = document.getElementById('logoutButton'); const postForm = document.getElementById('postForm'); const postContent = document.getElementById('postContent'); const counter = document.getElementById('counter'); const postsRoot = document.getElementById('posts'); const feedMessage = document.getElementById('feedMessage'); if (whoami) { whoami.textContent = currentUser.username || currentUser.email; } const renderPosts = (posts) => { if (!postsRoot) return; postsRoot.innerHTML = ''; posts.forEach((post) => { const article = document.createElement('article'); article.className = 'post-card'; const head = document.createElement('div'); head.className = 'post-head'; const avatar = document.createElement('div'); avatar.className = 'avatar-mini'; avatar.textContent = (post.author || 'F').charAt(0).toUpperCase(); const meta = document.createElement('div'); const strong = document.createElement('strong'); strong.textContent = post.author || 'Utilisateur'; const small = document.createElement('small'); small.textContent = post.time || 'À l\'instant'; meta.appendChild(strong); meta.appendChild(small); head.appendChild(avatar); head.appendChild(meta); const content = document.createElement('p'); content.textContent = post.content; article.appendChild(head); article.appendChild(content); postsRoot.appendChild(article); }); }; const fetchPosts = async () => { const posts = await window.Freshnet.getPosts(); renderPosts(posts || []); }; if (counter && postContent) { const updateCounter = () => { counter.textContent = `${postContent.value.length} / 500`; }; postContent.addEventListener('input', updateCounter); updateCounter(); } if (postForm && postContent) { postForm.addEventListener('submit', async (event) => { event.preventDefault(); const content = postContent.value.trim(); if (!content) { if (feedMessage) { feedMessage.textContent = 'Écris quelque chose avant de publier.'; feedMessage.className = 'message error'; } return; } try { await window.Freshnet.createPost(content); postContent.value = ''; if (counter) counter.textContent = '0 / 500'; if (feedMessage) { feedMessage.textContent = 'Publication ajoutée.'; feedMessage.className = 'message'; } await fetchPosts(); } catch (error) { if (feedMessage) { feedMessage.textContent = error.message || 'Impossible de publier ce message.'; feedMessage.className = 'message error'; } } }); } if (logoutButton) { logoutButton.addEventListener('click', async () => { await window.Freshnet.signOut(); window.location.href = 'connexion.html'; }); } await fetchPosts(); });
